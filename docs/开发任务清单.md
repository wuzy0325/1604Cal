```go
func (s *Service) ResolveAlarm(decision string) error {
    s.alarmMu.Lock()
    if !s.alarmPending {
        s.alarmMu.Unlock()
        return ErrNoPendingAlarm
    }
    s.alarmCh <- decision
    s.alarmPending = false
    s.alarmMu.Unlock()
    return nil
}
```

4. 在自动采集循环中处理决策:
```go

case AlarmDecisionRecollect:

case AlarmDecisionStop:
    // 停止采集
    return
}

func (c *Collector) Collect(ctx context.Context, channels []int, avgCount int) ([]float64, error) {
    if avgCount <= 1 {
        return c.driver.CollectData(ctx, channels)
    }
    
    // 多次采样
    samples := make([][]float64, avgCount)
    for i := 0; i < avgCount; i++ {
        data, err := c.driver.CollectData(ctx, channels)
        if err != nil {
            return nil, err
        }
        samples[i] = data
    }
    
    // 求平均
    result := make([]float64, len(channels))
    for ch := 0; ch < len(channels); ch++ {
        sum := 0.0
        for i := 0; i < avgCount; i++ {
            sum += samples[i][ch]
        }
        result[ch] = sum / float64(avgCount)
    }
    return result, nil
}
```

**验收标准:**
- [ ] `AverageCount > 1` 时执行多次采样
- [ ] 返回平均值

    n := float64(len(x))

    for i := range x {
        sumX += x[i]
        sumY += y[i]
        sumXY += x[i] * y[i]
        sumX2 += x[i] * x[i]
        sumY2 += y[i] * y[i]
    }
    slope = (n*sumXY - sumX*sumY) / (n*sumX2 - sumX*sumX)
